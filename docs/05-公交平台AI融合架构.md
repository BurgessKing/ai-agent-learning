# 05 — 公交平台 AI 融合架构

> **目标**：散包订单场景，用户通过"小途智能体"一句话完成报价→确认→下单
>
> **架构**：Sidecar（前端悬浮 AI 按钮）+ MCP（后端工具标准化）+ LangGraph（Agent 编排引擎）
>
> **LLM**：DeepSeek | **编排**：LangGraph Python | **Phase 1**：报价 + 下单

### 3.2 状态定义

```python
from typing import TypedDict, Annotated, Optional
import operator

class CharterAgentState(TypedDict):
    """散包 Agent 状态 — LangGraph 全局共享"""
    
    # 消息历史
    messages: Annotated[list, operator.add]
    
    # 当前阶段
    phase: str  # intent | extract | enrich | calculate | confirm | submit | summarize
    
    # 提取的参数
    customer_name: Optional[str]
    passenger_count: Optional[int]
    destination: Optional[str]
    travel_type: Optional[str]      # ONE_WAY / ROUND_TRIP / FULL_DAY
    vehicle_type: Optional[str]     # 大巴 / 中巴 / 商务车
    start_date: Optional[str]
    start_time: Optional[str]
    
    # 补全的信息
    route_distance_km: Optional[float]
    region: Optional[str]
    service_scope: Optional[str]
    supplier_id: Optional[int]
    
    # 报价结果
    quotation_amount: Optional[float]
    quotation_details: Optional[list]
    
    # 下单结果
    task_no: Optional[str]
    task_status: Optional[str]
    
    # 流程控制
    next_node: str
    error: Optional[str]
    user_confirmed: bool
```

### 3.3 节点代码骨架

```python
"""
散包 Agent — LangGraph 实现
技术栈: DeepSeek + LangGraph + MCP
"""

from langgraph.graph import StateGraph, END
from langgraph.prebuilt import ToolNode
from langchain_openai import ChatOpenAI
import httpx, json

# ===== LLM 配置 =====
llm = ChatOpenAI(
    model="deepseek-chat",
    base_url="https://api.deepseek.com/v1",
    temperature=0.3,
)

# ===== 节点实现 =====

def intent_node(state: CharterAgentState) -> dict:
    """Node 1: 意图识别 — 判断用户要做什么"""
    last_msg = state["messages"][-1]["content"]
    prompt = f"""判断用户意图，返回 JSON:
- pricing_request: 用户要计算包车报价
- order_request: 用户要下单
- modify_request: 用户要修改参数
- other: 其他

用户消息: {last_msg}

只返回 JSON: {{"intent": "...", "reason": "..."}}"""
    resp = llm.invoke([{"role": "user", "content": prompt}])
    intent_data = json.loads(resp.content)
    return {
        "phase": "extract" if intent_data["intent"] == "pricing_request" else "intent",
        "messages": [{"role": "assistant", "content": resp.content}]
    }

def extract_node(state: CharterAgentState) -> dict:
    """Node 2: 参数提取 — 从对话中提取 CharterTask 参数"""
    prompt = f"""从对话中提取包车订单参数，输出 JSON:

{{
  "customer_name": "企业名称或个人",
  "passenger_count": 人数,
  "destination": "目的地",
  "travel_type": "ONE_WAY|ROUND_TRIP|FULL_DAY",
  "vehicle_type": "大巴|中巴|商务车",
  "start_date": "YYYY-MM-DD",
  "start_time": "HH:mm"
}}

如果某个字段未提及，填 null。
缺失必要字段（客户名、人数、目的地、日期）时，设置 missing_fields 列出缺失项。

对话:
{format_messages(state['messages'])}"""
    
    resp = llm.invoke([{"role": "user", "content": prompt}])
    params = json.loads(resp.content)
    
    # 检查缺失字段
    missing = params.get("missing_fields", [])
    if missing:
        return {
            "phase": "extract",
            "next_node": "summarize",
            "messages": [{"role": "assistant", "content": f"请补充以下信息: {', '.join(missing)}"}]
        }
    
    return {
        "phase": "enrich",
        "customer_name": params.get("customer_name"),
        "passenger_count": params.get("passenger_count"),
        "destination": params.get("destination"),
        "travel_type": params.get("travel_type"),
        "vehicle_type": params.get("vehicle_type", "大巴"),
        "start_date": params.get("start_date"),
        "start_time": params.get("start_time", "08:00"),
    }

def enrich_node(state: CharterAgentState) -> dict:
    """Node 3: 信息补全 — 查路线距离、区域、车型"""
    # 通过 MCP 查询路线距离
    if not state.get("route_distance_km"):
        dist = call_mcp("route-mcp", "get_distance", {
            "destination": state["destination"]
        })
        state["route_distance_km"] = dist.get("distance_km")
    
    # 通过 MCP 查询区域信息
    if not state.get("region"):
        region = call_mcp("route-mcp", "get_region", {
            "destination": state["destination"]
        })
        state["region"] = region.get("code")
        state["region_name"] = region.get("name")
    
    return {
        "phase": "calculate",
        "route_distance_km": state.get("route_distance_km"),
        "region": state.get("region"),
        "region_name": state.get("region_name"),
    }

def calculate_node(state: CharterAgentState) -> dict:
    """Node 4: 报价计算 — 调 Admin quotation/calculate"""
    
    # 构建报价请求参数（对应 QuotationCalculateCmd）
    params = {
        "travelType": state["travel_type"],
        "serviceScope": state.get("service_scope", ""),
        "region": state["region"],
        "regionName": state.get("region_name", ""),
        "useDuration": estimate_duration(state["route_distance_km"]),
        "useMileage": state["route_distance_km"] * 1000 * (2 if state["travel_type"] == "ROUND_TRIP" else 1),
        "taskVehicles": [{
            "categoryId": get_vehicle_category(state["vehicle_type"]),
            "categoryName": state["vehicle_type"],
            "count": 1
        }]
    }
    
    # 调用 MCP → Admin quotation/calculate
    result = call_mcp("quotation-mcp", "calculate", params)
    
    if result.get("errors"):
        return {
            "phase": "summarize",
            "error": "; ".join(result["errors"]),
            "messages": [{"role": "assistant", "content": f"报价计算异常: {result['errors']}"}]
        }
    
    return {
        "phase": "confirm",
        "quotation_amount": result.get("totalAmount"),
        "quotation_details": result.get("basicDetails"),
        "messages": [{"role": "assistant", "content": format_quotation(result)}]
    }

def confirm_node(state: CharterAgentState) -> dict:
    """Node 5: 确认 — 等待用户确认下单"""
    last_msg = state["messages"][-1]["content"]
    
    # LLM 判断用户意图
    prompt = f"用户对报价¥{state['quotation_amount']}的回应是: '{last_msg}'。判断: confirm/cancel/modify。只返回一个词。"
    resp = llm.invoke([{"role": "user", "content": prompt}])
    decision = resp.content.strip()
    
    if decision == "confirm":
        return {"phase": "submit", "user_confirmed": True}
    elif decision == "cancel":
        return {"phase": "summarize", "next_node": END}
    else:
        return {"phase": "extract", "next_node": "extract"}  # 回到参数修改

def submit_node(state: CharterAgentState) -> dict:
    """Node 6: 下单 — 调 Admin charterTask/create"""
    
    params = {
        "customerType": "企业",
        "customerName": state["customer_name"],
        "passengerName": state["customer_name"],
        "passengerPhone": "待补充",  # 实际场景追问
        "passengerNum": state["passenger_count"],
        "travelType": state["travel_type"],
        "taskVehicles": [{
            "categoryId": get_vehicle_category(state["vehicle_type"]),
            "categoryName": state["vehicle_type"],
            "count": 1
        }],
        "goStartTime": f"{state['start_date']} {state['start_time']}:00",
        "goEndTime": f"{state['start_date']} 18:00:00",
        "useDuration": estimate_duration(state["route_distance_km"]),
        "estimatedAmount": state["quotation_amount"],
        "orderPlaceAmount": state["quotation_amount"],
        "supplierId": state.get("supplier_id", 1),
        "region": state["region"],
    }
    
    result = call_mcp("charter-mcp", "create", params)
    
    return {
        "phase": "summarize",
        "task_no": result.get("taskNo"),
        "task_status": "已创建",
    }

def summarize_node(state: CharterAgentState) -> dict:
    """Node 7: 汇总 — 返回最终结果给用户"""
    
    if state.get("task_no"):
        content = f"""
✅ 下单成功！

| 项目 | 内容 |
|------|------|
| 任务编号 | {state['task_no']} |
| 客户 | {state['customer_name']} |
| 行程 | {state['destination']} {state['travel_type']} |
| 人数 | {state['passenger_count']}人 |
| 车型 | {state['vehicle_type']} |
| 报价 | ¥{state['quotation_amount']} |
"""
    elif state.get("error"):
        content = f"❌ {state['error']}"
    else:
        content = f"💰 预估报价: ¥{state.get('quotation_amount', '—')}\n\n是否确认下单？"
    
    return {
        "messages": [{"role": "assistant", "content": content}],
        "next_node": END
    }


# ===== 条件路由 =====

def route_by_phase(state: CharterAgentState) -> str:
    """根据当前阶段决定下一步"""
    route_map = {
        "intent": "extract",
        "extract": "enrich" if state.get("customer_name") else "summarize",
        "enrich": "calculate",
        "calculate": "confirm",
        "confirm": "submit" if state.get("user_confirmed") else "summarize",
        "submit": "summarize",
        "summarize": END,
    }
    return route_map.get(state["phase"], END)


# ===== 构建 Graph =====

def build_charter_agent():
    graph = StateGraph(CharterAgentState)
    
    graph.add_node("intent", intent_node)
    graph.add_node("extract", extract_node)
    graph.add_node("enrich", enrich_node)
    graph.add_node("calculate", calculate_node)
    graph.add_node("confirm", confirm_node)
    graph.add_node("submit", submit_node)
    graph.add_node("summarize", summarize_node)
    
    graph.set_entry_point("intent")
    
    graph.add_conditional_edges("intent", route_by_phase)
    graph.add_conditional_edges("extract", route_by_phase)
    graph.add_conditional_edges("enrich", route_by_phase)
    graph.add_conditional_edges("calculate", route_by_phase)
    graph.add_conditional_edges("confirm", route_by_phase)
    graph.add_conditional_edges("submit", route_by_phase)
    graph.add_conditional_edges("summarize", route_by_phase)
    
    return graph.compile()
```

## 四、MCP Server 设计

### 4.1 与后端对接方式

```
MCP Server = 一个薄包装层，转发到 Admin API

方式: HTTP MCP Server（Python 进程，不嵌入 Java）
  Agent → MCP Client → HTTP POST → MCP Server → Admin API

Phase 1 简化: 不用完整 MCP 协议，用 FastAPI 做薄包装
  Agent → httpx → MCP 薄包装 (FastAPI) → Admin API
  即: 标准 REST 调用，MCP 协议在 Phase 3 补齐
```

### 4.2 quotation-mcp（报价）

"""

@app.post("/tools/calculate")

## 七、面试话术

> "我们采用了 Sidecar + MCP 架构。前端悬浮'小途智能体'按钮，Agent 服务基于 LangGraph 编排，后端业务接口通过 MCP 薄包装暴露。AI 不是替代后端，而是作为编排层——用户一句话，Agent 自动完成参数提取、信息补全、报价计算、确认下单的全流程。架构上前后端解耦，Agent 挂了不影响正常业务。"
